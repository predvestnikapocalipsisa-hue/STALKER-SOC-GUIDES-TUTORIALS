Для функций ниже будем использовать следующие обозначения:
	sobj - серверный объект
	obj - клиентский

по стори айди:
	local sobj = alife():story_object(14)
	local obj = level_object_by_sid(14)

по имени из all.spawn:
	local sobj = alife():object("esc_wolf")
	local obj = level_object_by_name("esc_wolf")

по имени нпс заспавненых через скрипт:
	local obj = level_object_by_spawn("esc_wolf")
	local sobj = alife_object_by_spawn("esc_wolf")

если вызов через диалог/логику то sobj можно определить так:
    local sobj = alife():object(obj:id())

Если вы увидите функции в которых фигурирует db.actor, но в аргументах (actor, obj), значит вместо db.actor можно указать obj и наоборот

function имя_функции(actor, obj)

Так же эти функции можно использовать в диалоге/логике и действия будут применяться к собеседникам или объектам, в кастом дате которых бала вызвана функция.

Также будут функции у которых определение через айди - можно указывать:
sobj.id или obj:id()

ОГСР от 3.432 получение левел и гейм вертекс по имени локации, кладем в _g.script 
function get_lv_gv(level_name, pos)
    local lg, ct
    if level.name() == level_name then
        lg = level_graph()
        ct = cross_table()
    else
        lg = level_graph(level_name)
        ct = cross_table(level_name)
    end

    local lv = lg:vertex_id(pos)
    if not lg:valid_vertex_id(lv) then
        lv = lg:nearest_vertex_id(pos)
    end
    local gv = ct:vertex(lv):game_vertex_id()
    return lv, gv
end

function transfer_items(victim, section, type)
	if db.actor and victim then
		if type == "in" then
			if victim:object(section) ~= nil then
				victim:transfer_item(victim:object(section), db.actor)
			end
		elseif type == "out" then
			if db.actor:object(section) ~= nil then
				db.actor:transfer_item(db.actor:object(section), victim)
			end
		end
		news_manager.relocate_item(db.actor, type, section)
	end
end
