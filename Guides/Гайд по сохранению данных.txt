Скрипты позволяют сохранять и загружать произвольные пользовательские данные. Это особенно полезно при создании квестов, где требуется хранить состояния NPC, настроек или других игровых флагов между сессиями.


Сохранение данных, которые переживают новую игру
Хранит числовые значения в именах файлов: varname_VALUE.sta 

Нюансы:
- В update каждый кадр не дёргать - держите значение в памяти, пишите по событию/таймеру.
- Если передать строку "abc", tonumber вернет nil, и value станет 0.

local SAVES_DIR  = "$game_saves$"
local FILE_MASK  = "%s_*.sta"       -- маска поиска: имяпеременной_число.sta
local FILE_TMPL  = "%s_%d.sta"      -- шаблон имени: имяпеременной_значение.sta

-- Санитизация имени: только буквы, цифры, подчёрки. Иначе FS может поехать.
local function sanitize(name)
    return (string.gsub(name, "[^%w_]", "_"))
end

-- Найти текущий файл переменной. Возвращает (полное_имя, значение) или (nil, nil).
local function find_file(name)
    local f = getFS()
    local safe = sanitize(name)
    local flist = f:file_list_open_ex(
        SAVES_DIR,
        bit_or(FS.FS_ListFiles, FS.FS_RootOnly),
        string.format(FILE_MASK, safe)
    )
    if flist:Size() == 0 then return nil, nil end

    local fname = flist:GetAt(0):NameFull()
    -- парсим: varname_123.sta → 123
    local _, _, val = string.find(fname, safe .. "_(%d+)%.sta")
    return fname, tonumber(val)
end

-- Запись значения. Переименовывает старый файл или создаёт новый.
function fs_save_var(name, value)
    value = math.floor(tonumber(value) or 0)
    local f = getFS()
    local safe = sanitize(name)
    local new_name = string.format(FILE_TMPL, safe, value)

    local old_name, _ = find_file(name)
    if old_name then
        -- переименовываем старый файл в новый
        f:file_rename(
            f:update_path(SAVES_DIR, old_name),
            f:update_path(SAVES_DIR, new_name),
            true
        )
    else
        -- файла нет — создаём пустой
        local fh = f:w_open(SAVES_DIR, new_name)
        if fh then f:w_close(fh) end
    end
    return value
end

-- Чтение значения. Если файла нет — возвращает default и НЕ создаёт файл.
function fs_load_var(name, default)
    local _, val = find_file(name)
    return val ~= nil and val or (default or 0)
end

-- Инкремент на delta (по умолчанию +1). Возвращает новое значение.
function fs_add_var(name, delta, default)
    local cur = fs_load_var(name, default or 0)
    return fs_save_var(name, cur + (delta or 1))
end


Использование:

-- Запись
fs_save_var("actor_deaths", 5)

-- Чтение
local deaths = fs_load_var("actor_deaths", 0)    -- вернёт 0 если нет

-- Счётчик (как в оригинале)
fs_add_var("actor_deaths")        -- +1, вернёт новое значение
fs_add_var("money_spent", 500)    -- +500

local file = "my_mod.ltx"

local function open(write)
    local ini = ini_file(file, true)
	if not ini then 
	ini_file:save_ini("gamedata\\config\\my_mod.ltx")
	end	
    if write then ini.readonly = false end
    return ini
end

local reader = {
    boolean = function(i,s,k) return i:r_bool(s,k)   end,
    number  = function(i,s,k) return i:r_float(s,k)  end,
    string  = function(i,s,k) return i:r_string(s,k) end,
}
local writer = {
    boolean = function(i,s,k,v) i:w_bool(s,k,v)   end,
    number  = function(i,s,k,v) i:w_float(s,k,v)  end,
    string  = function(i,s,k,v) i:w_string(s,k,v) end,
}

-- чтение конкретного параметра, default задаёт и тип, и fallback
function ltx_load_var(sect, key, default)
    local ini = open(false)
    if not ini:section_exist(sect) or not ini:line_exist(sect, key) then
        return default
    end
    return (reader[type(default)] or reader.string)(ini, sect, key)
end

function ltx_save_var(sect, key, value)
    local f = writer[type(value)]
    if not f then return false end
    local ini = open(true)
    if not ini:section_exist(sect) then ini:append_section(sect) end
    f(ini, sect, key, value)
    ini:save()
    return true
end

-- счётчик: прочитал, прибавил, записал
function ltx_add_var(sect, key, delta, default)
    local v = ltx_load_var(sect, key, default or 0) + (delta or 1)
    ltx_save_var(sect, key, v)
    return v
end




Сохранение в сейв

В ОГСР появилось нормально хранилище get_stored_vars()
Вот что написано на их вики:
Хранилище скриптовых переменных (SCRIPT_VARS_STORAGE)
Как использовать? Вот так вот:

local xvars = get_stored_vars()
xvars.test = 123
xvars.test2 = {}
xvars.test2.test1 = 456
Всё просто, использовать, как стандартную таблицу. Есть пара особенностей. Хранить лучше только простые типы: строки, числа, таблицы и т.п, в том числе вложенные. Объекты, вектора, время и т.п., то, что является userdata, лучше не использовать.

Вторая особенность - списки. Для прямого доступа к элементам списка ничего делать не нужно, но если захочется с этим списком работать, как с настоящим списком, т.к. итерировать, к примеру, или подсунуть его в unpack(...), то его нужно превратить в lua таблицу. Например:

xvars.test = { 1, 2, 3 }
local t = xvars.test()
for i, v in ipairs( t ) do
end
Операция выполнения, т.е. (), возвращает lua table, с которой уже можно делать, что угодно. По умолчанию, преобразование не рекурсивное, т.е. не затронет вложенные таблицы. Чтобы сделать его рекурсивным, нужно сделать вот так:

local t = xvars.test( true )
т.е. передать аргумент.

Хранилище от Artos - se_stor
🗃️ se_stor — Расширенное хранилище данных для S.T.A.L.K.E.R. (МОДУЛИ Artos(lua_helper+m_netpk+m_timers+se_stor))
se_stor — это продвинутая система хранения пользовательских данных в S.T.A.L.K.E.R., созданная сообществом (в частности по материалам, Malandrinus и xStream), и разработанная как альтернатива ограниченному pstor из xr_logic.script.

Она позволяет сохранять значительно больше данных, включая сложные типы (таблицы, векторы, временные метки), а также обходить ограничение в 8 КБ, присущее стандартному чанку актёра.

🔧 Основные возможности
✅ Поддерживаемые типы данных:
boolean → сохраняется как 1 байт
number → всегда как float (4 байта)
Для очень больших чисел рекомендуется преобразование в строку: ""..1234567890
string → как null-terminated C-строка
Макс. длина: ~8000–16000 байт (в зависимости от контекста)
table → только "простые" таблицы:
Без метатаблиц
Без рекурсии
Ключи и значения — только примитивы (number, string, boolean)
Автоматическая сериализация в строку (возможно с сжатием)
vector, CTime, binary chunk — специальные игровые типы
⚠️ Важно: при превышении лимита одного пакета (тоже ~8 КБ) сохранение упадёт — но se_stor может использовать несколько объектов-хранилищ, чтобы обойти это. 

🏗️ Архитектура
1. Специальный игровой объект CUST_ST (clsid.custom_storage)
2. Глобальная таблица storehouse
Все данные хранятся в общей таблице se_stor.storehouse
При сохранении — сериализуется и записывается в один или несколько custom_storage-объектов
При загрузке — восстанавливается автоматически

📥 Интерфейс API

➤ Запись
se_stor.set(var_name, value, f_check)
var_name — имя переменной (строка)
value — значение (любой поддерживаемый тип)
f_check — (опционально) флаг проверки размера перед записью
Чтобы удалить переменную — передайте nil как значение: 

se_stor.set("old_flag", nil)

➤ Чтение
local val = se_stor.get(var_name, default_value)
Если переменная не существует и default_value не задан — вернёт nil

➤ Удобные глобальные функции (в _G берутся из lua_helper.script)
delay_time = GetVar("noclip_delay_time", 30)  
SetVar("noclip_delay_time", noclip_mode.delay_time)

SetVar/GetVar — это алиасы на se_stor.set/se_stor.get

💡 Зачем это нужно?
Квестовые системы с сотнями флагов
Прогресс персонажа (скиллы, достижения, инвентарь вне рюкзака)
Динамические события, требующие хранения временных меток и состояний
Сложная экономика или репутационные системы
Например, если вы хотите сохранить список убитых монстров по типам, или карту открытых локаций, pstor быстро исчерпает лимит — а se_stor справится легко. 

✅ Вывод
se_stor — это must-have для серьёзных модов S.T.A.L.K.E.R., где стандартного pstor недостаточно. Он открывает двери для:

Глубокой персонализации персонажа
Сложных сюжетных ветвлений
Сохранения состояния мира между сессиями
При этом он остаётся совместимым с оригинальной архитектурой и использует те же механизмы сохранения, что и сама игра — просто "умнее".

Совет: если вы пишете мод с большим количеством состояний — начинайте с se_stor сразу. Перенос с pstor потом будет болезненным.

Далее пойдет инфа о pstor

🔹 Система pstor🔹
В S.T.A.L.K.E.R. реализована простая, но мощная система, которая интегрирована в xr_logic.script и используется через хранилище актора (db.storage).

🔹 Ray Kenkel: Небольшое хранилище важных данных, которые нужно сохранять, а не каждый раз пересоздавать. Условно говоря, если у НПС есть таймер, где описано что каждые 60 минут он обновит ассортимент, то без сохранения данных это бы значение сбрасывалось при перезагрузке. А так мы сохранили сколько уже осталось времени, а после загрузки вернули обратно это значение.

🔹 Главный нюанс. Ограничение в 8 КБ на чанк сохранения
Это жёсткое ограничение движка - вы не можете записать в один чанк более 8192 байт (8 КБ). Если попытаться превысить лимит при записи через packet:w_...() — игра аварийно завершится с ошибкой вида:
assertion failed: size < 8192

Как это работает под капотом?
Данные хранятся в db.storage[npc_id].pstor — это таблица Lua.
При сохранении игры (actor_binder:save) вызывается xr_logic.pstor_save_all(), который записывает все пары ключ-значение в бинарный поток.

А при загрузке сохранения (actor_binder:load) данные автоматически восстанавливаются:
xr_logic.pstor_load_all(self.object, reader)
Таким образом, вам не нужно вручную перезаписывать значения после загрузки — всё происходит автоматически.

Поддерживаются только три типа:
number - сохраняется как float
string - как нуль-терминированная строка
boolean - как булево значение

Функции для работы с ПСТОР

Сохранение значения
xr_logic.pstor_store(obj, "variable_name", value)

obj — объект, обычно db.actor (игрок).

"variable_name" — имя переменной в виде строки ей имя придумываем сами.

value — значение, может быть числом, строкой или булевым типом.


Пример: 
xr_logic.pstor_store(db.actor, "noclip_delay_time", 50)


Загрузка значения
local value = xr_logic.pstor_retrieve(obj, "variable_name", default_value)
Если переменная не найдена в сохранении — вернётся default_value.

Пример: 
local delay = xr_logic.pstor_retrieve(db.actor, "noclip_delay_time", 100)


-- прибавить к указанному счётчику актёра 1
xr_effects.inc_counter(db.actor,n,{"name"})	

-- отнять от указанного счётчика актёра 1 
xr_effects.dec_counter(db.actor,n,{"name"})	


Для более лёгкого использования вставьте функции показанные ниже, в конец файла \gamedata\scripts\_g.script

-- получить переменную 
function pstor_get_value(var_name, default_value)
  return xr_logic.pstor_retrieve(db.actor, var_name, default_value)
end

-- установить\сохранить переменную
function pstor_set_value(counter_name, value)
  xr_logic.pstor_store(db.actor, counter_name, value)
end

-- удалить\очистить переменную
function pstor_del_value(var_name)
  if not db.actor then
    abort("can't delete variable '%s', bacause actor is nil!", var_name)
  end
  local st = db.storage[db.actor:id()]
  if st.pstor and st.pstor[var_name] then
    st.pstor[var_name] = nil
  end
end

   return pstor_check_counter("fff", 10)
end

--работает аналогично, но без удаления
function pstor_check_counter_no_delete(name, cnt)
	if name and cnt then
		local c = xr_logic.pstor_retrieve(db.actor, name, 0)
		return c >= cnt
	else
		return false
	end
end

--' Обработчик счётчика, ставится в те места, где функция даст тру, тогда к счётчику будет прибавлено 1
function pstor_increment_counter(counter_name, counter_limit)
	local cnt = pstor_get_value(counter_name, 0) + 1
	if cnt >= counter_limit then
		pstor_del_value(counter_name)
    return true
	else
		pstor_set_value(counter_name, cnt)
    return false
	end
end


Ещё с оргинала в \gamedata\scripts\xr_effects.script есть функции inc_counter и dec_counter, прибавить к указанному счётчику актёра 1 или отнять

=inc_counter(имя_счетчика)
создает счетчик и добавляет 1

=dec_counter(имя_счетчика)
создает счетчик и отнимает 1

[logic]
active = nil
on_death = death

[death]
on_info = %=inc_counter(fff)%

в \gamedata\scripts\xr_conditions.script есть counter_greater у него надо набить на 1 больше чем указано, такой нюанс

=counter_greater(имя_счетчика:число)
возвращает true, если значение указанного счётчика БОЛЬШЕ указанного числа

Используются только в логике:

в этом примере показана логика рестриктора, который проверяет значение счётчика, при выполнении вызывает функцию спавн_нпс
[logic]
active = sr_idle

[sr_idle]
on_info = {=counter_greater(my_counter:5)} %=spawn_npc%
