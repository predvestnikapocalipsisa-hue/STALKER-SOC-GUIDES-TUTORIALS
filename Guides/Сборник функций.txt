-- НПС передает предмет нам, либо мы ему передаем, при этом у предмет исчезает.
function relocate_item_my(victim, section, type)
	if db.actor and victim then
		if type == "in" then
			if victim:object(section) ~= nil then
				victim:transfer_item(victim:object(section), db.actor)
			end
		elseif type == "out" then
			db.actor:transfer_item(db.actor:object(section), victim)
		end
		news_manager.relocate_item(db.actor, type, section)
	end
end

-- ОГСР от 3.432
-- получение левел и гейм вертекс по имени локации, кладем в _g.script 
function get_lv_gv(level_name, pos)
    local lg, ct
    if level.name() == level_name then
        lg = level_graph()
        ct = cross_table()
    else
        lg = level_graph(level_name)
        ct = cross_table(level_name)
    end

    local lv = lg:vertex_id(pos)
    if not lg:valid_vertex_id(lv) then
        lv = lg:nearest_vertex_id(pos)
    end
    local gv = ct:vertex(lv):game_vertex_id()
    return lv, gv
end

--------------------------------------------------------------------------------------------------------------
-- Новый спавн только по координатам, кладем в _g.script  
function alife_create(section, level_name, pos)
    local lv, gv = get_lv_gv(level_name, pos)
    return alife():create(section, pos, lv, gv)
end

function esc_timer()
	local iTimer = time_global() + 10*1000
	local function check_timer()
			 return time_global() > iTimer
		  end
	level.add_call(check_timer, kill_npc)
end

-- смена визуала нпс через диалог/логику на ОГСР
function esc_timer(actor, obj)
local iTimer = time_global() + 1
local sobj = alife():object(obj:id())
   if xr_logic.pstor_retrieve(db.actor, "set_switch_offline"..obj:id(), 0) ~= 1 then
      sobj.visual_name = "actors\\neytral\\stalker_neytral_rukzak_3" --визуал
      alife():set_switch_online(sobj.id, false)
      xr_logic.pstor_store(db.actor, "set_switch_offline"..obj:id(), 1)
   end
   local function check_timer()
      return time_global() > iTimer
   end
   local function switch_online()
      if xr_logic.pstor_retrieve(db.actor, "set_switch_online"..sobj.id, 0) ~= 1 then
         alife():set_switch_online(sobj.id, true)
         xr_logic.pstor_store(db.actor, "set_switch_online"..sobj.id, 1)
      end
   end
   level.add_call(check_timer, switch_online)           
end
