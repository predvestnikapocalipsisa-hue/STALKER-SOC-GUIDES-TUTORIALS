Мб кому полезно будет, созадаем файл .bat указываем внутри свой путь до папки levels и скрипт удалит все файлы кроме .gct .spawn .ai .game .graph

@echo off
setlocal enabledelayedexpansion

REM Путь к основной папке levels
set "root=C:\STALKER\S.T.A.L.K.E.R.1.0006\gamedata\levels"

REM Расширения, которые НЕ нужно удалять
set "keep_exts=.gct .spawn .ai .game .graph"

REM Папки, которые нужно удалить
set "del_dirs=terrain meshes"

echo Поиск и обработка всех подпапок в: %root%
echo.

REM Перебор всех подпапок внутри levels
for /d %%D in ("%root%\*") do (
    echo Обработка папки: %%~nxD
    pushd "%%D"
    
    REM 1. Удаление всех файлов, кроме нужных расширений
    echo Удаление лишних файлов...
    for %%A in (*) do (
        set "ext=%%~xA"
        set "match=no"
        for %%E in (%keep_exts%) do if /i "%%E"=="!ext!" set "match=yes"
        if "!match!"=="no" (
            echo Удаление файла: %%A
            del /q "%%A" >nul 2>&1
            if exist "%%A" echo [!] Не удалось удалить: %%A
        )
    )

    REM 2. Удаление папок terrain и meshes
    for %%K in (%del_dirs%) do (
        if exist "%%K" (
            echo Удаление папки: %%K
            rd /s /q "%%K" >nul 2>&1
            if exist "%%K" echo [!] Не удалось удалить папку: %%K
        )
    )

    popd
)

echo.
echo Очистка завершена.
pause

Гайд как запускать несколько модов с одной папки, разные fsgame.ltx, выглядит это так,  из-за того что геймдаты разные и дб надо переименовывать

Весь смысл в том, чтобы использовались одни и те же  игровые архивы


Принцип работы такой, запустили батник ОРИГ и дб  архивы переименовались в gamedata_orig.db*, иначе геймдата не подхватывается..

Запускаем потом ОГСР и архивы снова переименовываются в gamedata_ogsr.db* и так до бесконечности ♾️ при каждом запуске, происходит это мгновенно ⚙️

Сделать нужно два батника, первый для ОГСР
@echo off
REM Чистка логов
REM rd /s /q "%~dp0appdata\logs"

REM Проверка и переименование fsgame.ltx
if exist "fsgame.ltx_ogsr" (
    ren "fsgame.ltx" "fsgame.ltx_orig"
    ren "fsgame.ltx_ogsr" "fsgame.ltx"
)

@echo off
:: Восстановить обычные файлы из _orig (если есть)
for %%f in (gamedata_orig.db*) do (
    if exist "%%f" ren "%%f" "gamedata.db%%~xf"

:: Применить ogsr-версию
for %%f in (gamedata.db*) do (
    if exist "%%f" ren "%%f" "gamedata_ogsr%%~xf"
)

Открой папку с файлами
В адресной строке проводника (сверху) введи powershell и нажми Enter — откроется окно PowerShell в этой папке.
Скопируй и вставь эту команду целиком:

Добавить строку ко всем файлам в папке

Get-ChildItem *.script | ForEach-Object {
    $content = Get-Content $_.FullName -Raw
    if (-not $content.StartsWith("-- -*- mode: lua; coding: windows-1251-dos -*-")) {
        Set-Content $_.FullName ("-- -*- mode: lua; coding: windows-1251-dos -*-`n" + $content)
    }
}

Меняет кодировку с кириллицы 1251 на UTF8

Get-ChildItem *.script | ForEach-Object {
    $content = Get-Content $_.FullName
    $utf8NoBom = New-Object System.Text.UTF8Encoding($false)
    [System.IO.File]::WriteAllLines($_.FullName, $content, $utf8NoBom)
}

для папки текст\рус
Get-ChildItem *.xml | ForEach-Object {
    $content = Get-Content $_.FullName -Encoding Default
    Set-Content $_.FullName -Value $content -Encoding UTF8
}

потом заменить *.xml на *.ltx (эта команда учитывает подпапки)

Get-ChildItem -Recurse -Filter *.xml | ForEach-Object {
    $filePath = $_.FullName
    # Пробуем прочитать как UTF-8
    try { $utf = [System.IO.File]::ReadAllText($filePath, [System.Text.UTF8Encoding]::new($false)) } catch { $utf = $null }
    # Пробуем прочитать как ANSI (1251)
    $ansi = [System.IO.File]::ReadAllText($filePath, [System.Text.Encoding]::GetEncoding(1251))

    # Если UTF-8 прочитался НЕ как кракозябры (есть буквы "а-яА-Я"), оставляем как есть
    if ($utf -match '[а-яА-ЯёЁ]') {
        # Уже UTF-8 — ничего не делать
        Write-Host "⏩ Уже UTF-8: $($_.Name)" -ForegroundColor Cyan
    }
    # Иначе — это ANSI, конвертируем в UTF-8 без BOM
    else {
        $utf8NoBom = New-Object System.Text.UTF8Encoding($false)
        [System.IO.File]::WriteAllText($filePath, $ansi, $utf8NoBom)
        Write-Host "✅ Перекодирован: $($_.Name)" -ForegroundColor Green
    }
}
