1. Базовые функции

Для функций ниже будем использовать следующие обозначения: sobj - серверный объект, obj - клиентский.

по стори айди:

local sobj = alife():story_object(14)
local obj = level_object_by_sid(14)

по имени из all.spawn:

local sobj = alife():object("esc_wolf")
local obj = level_object_by_name("esc_wolf")

по имени нпс заспавненых через скрипт:

local obj = level_object_by_spawn("esc_wolf")
local sobj = alife_object_by_spawn("esc_wolf")

если вызов через диалог/логику то sobj можно определить так:

local sobj = alife():object(obj:id())

Если вы увидите функции в которых фигурирует db.actor и в аргументах стоит (actor, obj), значит вместо db.actor можно указать obj и наоборот

function имя_функции(actor, obj)

Так же эти функции можно использовать в диалоге/логике и действия будут применяться к собеседникам или объектам, в кастом дате которых бала вызвана функция.

Также будут функции у которых определение через айди - можно указывать:

sobj.id или obj:id()

Нужно в _g.script добавить функции:

function level_object_by_name( name )

2. Координаты

get_lv_gv

ОГСР от 3.432 получение левел и гейм вертекс по имени локации, кладем в _g.script.

function get_lv_gv(level_name, pos)
    local lg, ct
    if level.name() == level_name then
        lg = level_graph()
        ct = cross_table()
    else
        lg = level_graph(level_name)
        ct = cross_table(level_name)
    end

    local lv = lg:vertex_id(pos)
    if not lg:valid_vertex_id(lv) then
        lv = lg:nearest_vertex_id(pos)
    end
    local gv = ct:vertex(lv):game_vertex_id()
    return lv, gv
end

Снятие координат (F1)

Находясь в меню можно будет жать Ф1 и появятся коорды. Откройте файл \gamedata\scripts\ui_main_menu.script. Через поиск найдите function main_menu:OnKeyboard(dik, keyboard_action). В этот участок if keyboard_action == ui_events.WINDOW_KEY_PRESSED then добавим это:

if dik == DIK_keys.DIK_F1 then
    local gv = db.actor:game_vertex_id()
    local lv = db.actor:level_vertex_id()
    local pos = db.actor:position()
    local posx = pos.x
    local posy = pos.y
    local posz = pos.z
    local dir = db.actor:direction()
    local dirx = dir.x
    local diry = dir.y
    local dirz = dir.z
    news_manager.send_tip(db.actor, "Position:".."\\n".."X: "..posx.."\\n".."Y: "..posy.."\\n".."Z: "..posz.."\\n".."Direction: ".."\\n".."X: "..dirx.."\\n".."Y: "..diry.."\\n".."Z: "..dirz.."\\n".."Game Vertex ID: "..gv.."\\n".."Level Vertex ID: "..lv.."\\n".."That's all.", nil, nil, 100000)    
    console:execute("main_menu off")
end

transfer_items

НПС передает предмет нам, либо мы ему передаем, при этом у нас предмет исчезает. "in" - нпс отдает предмет нам, "out" - мы отдаем предмет нпс.

function transfer_items(victim, section, type)
    if db.actor and victim then
        if type == "in" then
            if victim:object(section) ~= nil then
                victim:transfer_item(victim:object(section), db.actor)
            end
        elseif type == "out" then
            if db.actor:object(section) ~= nil then
                db.actor:transfer_item(db.actor:object(section), victim)
            end
        end
        news_manager.relocate_item(db.actor, type, section)
    end
end

Пример вызова:
function my(actor, obj)
    Имя_файла.transfer_items(obj, "wpn_rpg7", "in")
end

clear_inv

Очистка инвентаря гг от всех предметов (если применять для нпс, тогда нужно делать исключение для болта, иначе вылет).

local keep_item = {
    ["bolt"] = true,
    ["device_pda"] = true,
    ["wpn_knife"] = true,
}

function clear_inv()
    if not db.actor then return end
    
    local to_remove = {}
    db.actor:iterate_inventory(function(_, item)
        local section = item:section()
        if not keep_item[section] then
            table.insert(to_remove, item:id())
        end
    end, nil)
    
    for _, id in ipairs(to_remove) do
        alife():release(alife():object(id), true)
    end
end

spawn_item_anywhere

Рандомное место спавна по всей зоне или по указанной локации.

local gv_cache = {}

local function all_gvs()
    if gv_cache then return gv_cache end
    local t, g, i = {}, game_graph(), 0
    while g:valid_vertex_id(i) do
        t[#t+1] = i
        i = i + 1
    end
    gv_cache = t
    return t
end

local function level_gvs(lname)
    if gv_cache[lname] then return gv_cache[lname] end
    local t, g, i = {}, game_graph(), 0
    while g:valid_vertex_id(i) do
        if alife():level_name(g:vertex(i):level_id()) == lname then
            t[#t+1] = i
        end
        i = i + 1
    end
    gv_cache[lname] = t
    return t
end

function spawn_item_anywhere(section, lname)
    lname = lname or alife():level_name(alife():level_id())
    local t = lname and level_gvs(lname) or all_gvs()
    if #t == 0 then return nil end

    local gv = t[math.random(#t)]
    local v  = game_graph():vertex(gv)
    local s = alife():create(section, v:level_point(), v:level_vertex_id(), gv)
    level.map_add_object_spot_ser(s.id, "green_location", s:name())
end

Взаимодействие с параметрами текущей погоды. Взаимодействовать скриптом можно только с параметрами которые описаны в CEnvDescriptor в lua_help.

local env = level.environment()
local env_desc = env:getCurrentWeather( idx )  
env_desc.wind_velocity = factor

Это загрузит секции погоды в Current[0] и Current[1]. Далее погода будет плавно меняться от Current[0] к Current[1].
level.set_weather( weather, true )
