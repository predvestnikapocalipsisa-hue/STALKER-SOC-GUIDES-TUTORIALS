Мб кому полезно будет, созадаем файл .bat указываем внутри свой путь до папки levels и скрипт удалит все файлы кроме .gct .spawn .ai .game .graph

@echo off
setlocal enabledelayedexpansion

REM Путь к основной папке levels
set "root=C:\STALKER\S.T.A.L.K.E.R.1.0006\gamedata\levels"

REM Расширения, которые НЕ нужно удалять
set "keep_exts=.gct .spawn .ai .game .graph"

REM Папки, которые нужно удалить
set "del_dirs=terrain meshes"

echo Поиск и обработка всех подпапок в: %root%
echo.

REM Перебор всех подпапок внутри levels
for /d %%D in ("%root%\*") do (
    echo Обработка папки: %%~nxD
    pushd "%%D"
    
    REM 1. Удаление всех файлов, кроме нужных расширений
    echo Удаление лишних файлов...
    for %%A in (*) do (
        set "ext=%%~xA"
        set "match=no"
        for %%E in (%keep_exts%) do if /i "%%E"=="!ext!" set "match=yes"
        if "!match!"=="no" (
            echo Удаление файла: %%A
            del /q "%%A" >nul 2>&1
            if exist "%%A" echo [!] Не удалось удалить: %%A
        )
    )

    REM 2. Удаление папок terrain и meshes
    for %%K in (%del_dirs%) do (
        if exist "%%K" (
            echo Удаление папки: %%K
            rd /s /q "%%K" >nul 2>&1
            if exist "%%K" echo [!] Не удалось удалить папку: %%K
        )
    )

    popd
)

echo.
echo Очистка завершена.
pause
