local jump_artefact = "af_medusa"

function on_jump_handler()
	if db.actor then
		local target_item = nil
			db.actor:iterate_belt(function(owner, item)  
				if item and item:section() == jump_artefact then
					target_item = item
				end
			end)
		if target_item then
			local fx_particle = particles_object([[anomaly2\gravity_blast_01]])
				fx_particle:play_at_pos(db.actor:position())
			local snd = sound_object([[anomaly\anomaly_gravy_blast01]])
				snd: play_no_feedback(db.actor,sound_object.s2d, 0, vector():set(0, 0, 0), 1.0)
			local sobj = alife():object(target_item:id())
			if sobj then
				alife():release(sobj, true)
			end
		end
	end
end

То есть если артефакт(в моем случае медуза) весит на поясе артов и если при этом прыгнуть, то арт удаляется + возникает эффект грави аномалии и звук аномалии.

-----------------------------ДЛЯ OGSR И ТЧ-Счетчик предметов в диалоговом окне---------------------------=
----------------------------------------------------------------------------------------------------------

В ваниле, когда в диалоговом окне например тебе дают 5 пачек 9х18, и тебе 5 паз присылается новость о получении предмета(пачки 9х18).
Я сделал иначе: теперь когда ты получаешь несколько предметов, тебе приходит одна новость с названием предмета и его количеством.

в dialogs.script нужно добавить(или заменить за место функции "relocate_item_section", как угодно):

function relocate_n_items_section(victim, section, type, count)
    if db.actor and victim then
          if type == "in" then
        for i=1,count do
                         alife():create(section,     
                             db.actor:position(),
                                   db.actor:level_vertex_id(),     
                                   db.actor:game_vertex_id(),
                                   db.actor:id())
        end
    elseif type == "out" then
      db.actor:iterate_inventory(function (dummy,item)
        if item:section()==section and count>0 then 
          db.actor:transfer_item(item, victim)
          count=count-1
        end
       end, nil)
          end
          news_manager.relocate_item(db.actor, type, section, count)
     end
end

victim - это first_speaker или second_speaker
section - сама секция предмета
type - есть 2 типа: "in"(актор забирает предмет) и "out"(актор отдает предмет)
count - число выдаваемых/забираемых предметов


потом в news_manager.script поменять старый relocate_item на этот:

function relocate_item(actor, type, item, count)
	if db.actor == nil then return false end
	if not count then count = 1 end
	--' Играем дефолтный звук
	if type == "in" then
		local task_texture, task_rect = get_texture_info("ui_iconsTotal_found_thing")
		local news_text = "%c[255,160,160,160]"..game.translate_string("general_in_item").."\\n".."%c[default]"..iif(count>1,count.."x ","")..game.translate_string(get_inv_name(item))		
		if db.actor:is_talking() then
			db.actor:give_talk_message(news_text, task_texture, task_rect,"iconed_answer_item")
		else
			db.actor:give_game_news(news_text, task_texture, task_rect, 0, 3000)
		end		
	elseif type == "out" then
		local task_texture, task_rect = get_texture_info("ui_iconsTotal_lost_thing")
		local news_text = "%c[255,160,160,160]"..game.translate_string("general_out_item").."\\n".."%c[default]"..iif(count>1,count.."x ","")..game.translate_string(get_inv_name(item))		
		if db.actor:is_talking() then
			db.actor:give_talk_message(news_text, task_texture, task_rect,"iconed_answer_item")
		else
			db.actor:give_game_news(news_text, task_texture, task_rect, 0, 3000)
		end		
	end
end
