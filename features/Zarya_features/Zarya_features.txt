================================ создавание предметов в items.ltx ===============================================================

Разберем дефолтное создавание предметов в игре, навык простой но очень важный...

У каждого предмета есть свой класс, который описан в двигле игры 

вот самые базовые классы:

II_ATTCH
II_FOOD
II_MEDKI

класс влияет на текст при нажатии на правую кнопки мыши по предмету. 

II_ATTCH - это просто предмет, вы можете использовать его в скриптах и ТД. Но в игре с ним нельзя никак взаимодействовать
II_FOOD - применяемый предмет, при нажатии правой кнопкой мыши появляется текст "Сьесть"
II_MEDKI - то же самое что и II_FOOD, но вместо "сьесть" - "использовать"

================================ опыт с занесением в таблицу значений скриптом ===============================================================

Недавно задрочился с таблицами и прочей шляпой.
И готов показать небольшие функции которые сделал.

Для начала, необходимо создать таблицу:

local items = {} -- итоговая таблица

затем, когда нам надо будет в нее что-то записать. Делаем так:

        local name = blabla
		local item = belele
		items[name] = item

тем самым, таблица будет выглядеть как то так:

local items = 
{
	blabla = belele
} 

Теперь же, если нам нужно узнать че за item по name, то делаем так:

	local obj = items[blabla]

и все, все готово.

function deathStatCollect()
  if not game_options.death_stat then return end
  
  local f = getFS()
  amk.send_tip("%c[255,0,255,0]«афиксирована смерть сталкера по кличке ћеченый.%c[default]", "Ћќ јЋ№Ќјя —≈“№", 0, 15, "gen_info")
  local flist = f:file_list_open_ex("$game_saves$", bit_or(FS.FS_ListFiles, FS.FS_RootOnly), "death_*.sta")
  local f_cnt = flist:Size()

  if f_cnt > 0 then
    local file = flist:GetAt(0)
    local fileName = file:NameFull()
    --amk.send_tip(fileName, "”борщик", 0, 15, "gen_info")
    local _first = string.find(fileName, "_", 1, true)
    local _second = string.find(fileName, "_", _first + 1, true)

    local lastDate = string.sub(fileName, _first + 1, _second - 1)
    --amk.send_tip(lastDate, "”борщик", 0, 15, "gen_info")

    local deathCount = string.sub(fileName, _second + 1, string.find(fileName, ".", 1, true) - 1)
    --amk.send_tip(deathCount, "”борщик", 0, 15, "gen_info")

    local fileNameNew = "death_"..lastDate.."_"..tostring(deathCount + 1)..".sta"
    local f1 = f:update_path("$game_saves$", fileName)
    local f2 = f:update_path("$game_saves$", fileNameNew)
    --amk.send_tip(f1, "”борщик", 0, 15, "gen_info")
    --amk.send_tip(f2, "”борщик", 0, 15, "gen_info")
    f:file_rename(f1, f2, true)
  else-- не актуально так как файл будет создан копированием при старте новой игры, но пусть будет :)
    --amk.send_tip("√√ помер первый раз", "”борщик", 0, 15, "gen_info")
    
    local f1 = f:update_path("$game_config$", "narod_opt.ltx")

    local f2 = f:update_path("$game_saves$", "death_"..lastDate.."_1.sta")
    
    f:file_copy(f1, f2)
  end
